test("Nocturnal house rules: a roofed exhibit turns a hedgehog's clock round, and the measurement counts the roof", async () => {
  const N = await import("../packs/BP/scripts/animals/nightRules.js");
  assert.equal(N.sleepNow(false, false), true, "by day, outside a nocturnal house: asleep");
  assert.equal(N.sleepNow(true, false), false, "at night: up");
  assert.equal(N.sleepNow(false, true), false, "in a nocturnal house by day: up");
  assert.equal(N.sleepNow(true, true), true, "in a nocturnal house at night: asleep");
  assert.equal(N.roofNeeded(100, BALANCE), 90);
  assert.equal(N.isNocturnalHouse({ usableArea: 100, roofedCells: 90 }, BALANCE), true);
  assert.equal(N.isNocturnalHouse({ usableArea: 100, roofedCells: 89 }, BALANCE), false);
  assert.equal(N.isNocturnalHouse({ usableArea: 100 }, BALANCE), false, "not measured for a roof yet");
  assert.match(N.nocturnalLine({ usableArea: 100, roofedCells: 95 }, "Hedgehogs", BALANCE), /yes.*95 of 100 blocks roofed.*hedgehogs are up by day/);
  assert.match(N.nocturnalLine({ usableArea: 100, roofedCells: 40 }, "Hedgehogs", BALANCE), /no \(40 of 90 blocks roofed.*sleep by day/);
  assert.match(N.stateText(true, false, false), /asleep for the day/);
  assert.match(N.stateText(true, false, true), /asleep for the night/);
  assert.match(N.stateText(false, true, false), /curled into a ball/);
  assert.match(N.stateText(false, false, false), /awake and foraging/);
  // The measurement: a 9 x 9 pen walled with barriers, a stone roof 5 up over all but the east row,
  // which only has leaves over it.
  const grid = (roofAll) => (x, y, z) => {
    if (x < -5 || x > 5 || z < -5 || z > 5) return "air";
    if (x === -5 || x === 5 || z === -5 || z === 5) return y >= 0 && y <= 4 ? "barrier" : y < 0 ? "solid" : "air";
    if (y < 0) return "solid";
    if (y === 5 && (roofAll || x < 4)) return "solid";
    if (y === 7 && x === 4) return "leaves";
    return "air";
  };
  const run = (g, roofScan) => {
    const r = newResult({ x: 0, y: 0, z: 0 });
    const it = floodTerrestrial(g, { x: 0, y: 0, z: 0 }, { halfWidth: 32, halfHeight: 16, maxDrop: 3, minHeight: 3, cellsPerYield: 1e9, roofScan }, r);
    while (!it.next().done);
    return r;
  };
  const part = run(grid(false), 24);
  assert.equal(part.usableArea, 81);
  assert.equal(part.roofedCells, 72, "the east row is under leaves, which are not a roof");
  assert.equal(N.isNocturnalHouse(part, BALANCE), false, "72 of 81: not quite roofed over");
  const all = run(grid(true), 24);
  assert.equal(all.roofedCells, 81);
  assert.equal(N.isNocturnalHouse(all, BALANCE), true);
  assert.equal(run(grid(true), 0).roofedCells, 0, "only counted when asked");
});
